<script>
    // FIREBASE INITIALIZATION
    const firebaseConfig = {
        databaseURL: "https://hahahah-4f65c-default-rtdb.firebaseio.com/"
    };
    firebase.initializeApp(firebaseConfig);
    const db = firebase.database();

    // Giriş məlumatları
    const USER = "K9ESCOBAR211_";
    const PASS = "AA555_67";

    function login() {
        let u = document.getElementById('admin-user').value.trim();
        let p = document.getElementById('admin-pass').value.trim();

        if(u === USER && p === PASS) {
            document.getElementById('login-screen').style.display = 'none';
            document.getElementById('dashboard-screen').style.display = 'block';
            loadData();
        } else {
            alert('İstifadəçi adı və ya şifrə yanlışdır!');
        }
    }

    function logout() {
        document.getElementById('login-screen').style.display = 'block';
        document.getElementById('dashboard-screen').style.display = 'none';
    }

    function loadData() {
        db.ref('kredit_muraciətləri').on('value', (snapshot) => {
            let data = snapshot.val();
            let tbody = document.getElementById('app-list');
            tbody.innerHTML = '';

            if(!data) {
                tbody.innerHTML = '<tr><td colspan="5" style="text-align:center;">Heç bir müraciət tapılmadı.</td></tr>';
                return;
            }

            let total = 0, approved = 0, rejected = 0;

            Object.keys(data).reverse().forEach(key => {
                total++;
                let item = data[key];

                if(item.status === 'Təsdiqləndi') approved++;
                if(item.status === 'İmtina edildi') rejected++;

                let badgeClass = 'badge-pending';
                if(item.status === 'Təsdiqləndi') badgeClass = 'badge-approved';
                if(item.status === 'İmtina edildi') badgeClass = 'badge-rejected';

                let row = `
                    <tr>
                        <td>
                            <strong>${item.fullname || '-'}</strong><br>
                            <small style="color:#94a3b8">FIN: ${item.fin || '-'}</small><br>
                            <small style="color:#38bdf8">${item.phone || '-'}</small>
                        </td>
                        <td>
                            <b>${item.amount} AZN</b> / ${item.months} Ay<br>
                            <small style="color:#94a3b8">Aylıq: ${item.monthly} AZN</small>
                        </td>
                        <td>
                            <div class="rel-box">
                                <b>${item.rel1 ? item.rel1.type : 'Qohum'}:</b> ${item.rel1 ? item.rel1.name : '-'} (${item.rel1 ? item.rel1.phone : '-'})
                            </div>
                            <div class="rel-box">
                                <b>${item.rel2 ? item.rel2.type : 'Dost'}:</b> ${item.rel2 ? item.rel2.name : '-'} (${item.rel2 ? item.rel2.phone : '-'})
                            </div>
                        </td>
                        <td>
                            <span class="badge ${badgeClass}">${item.status}</span>
                        </td>
                        <td>
                            <button class="btn-action btn-approve" onclick="updateStatus('${key}', 'Təsdiqləndi')">Təsdiqlə</button>
                            <button class="btn-action btn-reject" onclick="updateStatus('${key}', 'İmtina edildi')">İmtina Et</button>
                        </td>
                    </tr>
                `;
                tbody.innerHTML += row;
            });

            document.getElementById('stat-total').innerText = total;
            document.getElementById('stat-approved').innerText = approved;
            document.getElementById('stat-rejected').innerText = rejected;
        });
    }

    function updateStatus(id, newStatus) {
        db.ref('kredit_muraciətləri/' + id).update({
            status: newStatus
        });
    }
</script>
