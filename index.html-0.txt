<script>
    // FIREBASE INITIALIZATION
    const firebaseConfig = {
        databaseURL: "https://hahahah-4f65c-default-rtdb.firebaseio.com/"
    };
    firebase.initializeApp(firebaseConfig);
    const db = firebase.database();

    let currentAppId = null;

    function updateCalc() {
        let amount = parseInt(document.getElementById('amount-slider').value);
        let months = parseInt(document.getElementById('month-slider').value);

        document.getElementById('amount-val').innerText = amount + " AZN";
        document.getElementById('month-val').innerText = months + " Ay";

        let r = (14 / 100) / 12;
        let monthly = (amount * (r * Math.pow(1 + r, months)) / (Math.pow(1 + r, months) - 1)).toFixed(2);
        let total = (monthly * months).toFixed(2);

        document.getElementById('calc-monthly').innerText = monthly;
        document.getElementById('calc-total').innerText = total;
    }

    function goToStep(stepNum) {
        document.querySelectorAll('.step-content').forEach(el => el.classList.remove('active'));
        document.getElementById('step' + stepNum).classList.add('active');

        for(let i = 1; i <= 4; i++) {
            let ind = document.getElementById('ind-step' + i);
            ind.classList.remove('active', 'completed');
            if(i === stepNum) ind.classList.add('active');
            else if(i < stepNum) ind.classList.add('completed');
        }
    }

    function validateAndGoStep3() {
        let fullname = document.getElementById('cust-fullname').value.trim();
        let fin = document.getElementById('cust-fin').value.trim();
        let phone = document.getElementById('cust-phone').value.trim();

        if(!fullname || !fin || !phone) {
            alert('Zəhmət olmasa bütün şəxsi məlumatları doldurun!');
            return;
        }
        goToStep(3);
    }

    function submitApplication() {
        let rel1Name = document.getElementById('rel1-name').value.trim();
        let rel1Phone = document.getElementById('rel1-phone').value.trim();
        let rel2Name = document.getElementById('rel2-name').value.trim();
        let rel2Phone = document.getElementById('rel2-phone').value.trim();

        if(!rel1Name || !rel1Phone || !rel2Name || !rel2Phone) {
            alert('Zəhmət olmasa qohum və dost/tanış məlumatlarını tam daxil edin!');
            return;
        }

        goToStep(4);

        let amount = document.getElementById('amount-slider').value;
        let months = document.getElementById('month-slider').value;
        let monthly = document.getElementById('calc-monthly').innerText;
        let fullname = document.getElementById('cust-fullname').value;
        let fin = document.getElementById('cust-fin').value.toUpperCase();
        let phone = document.getElementById('cust-phone').value;

        let rel1Type = document.getElementById('rel1-type').value;
        let rel2Type = document.getElementById('rel2-type').value;

        let appData = {
            fullname: fullname,
            fin: fin,
            phone: phone,
            amount: amount,
            months: months,
            monthly: monthly,
            rel1: { type: rel1Type, name: rel1Name, phone: rel1Phone },
            rel2: { type: rel2Type, name: rel2Name, phone: rel2Phone },
            status: "Şərtlər yoxlanılır",
            timestamp: new Date().toISOString()
        };

        // Firebase-ə yazılır
        let newRef = db.ref('kredit_muraciətləri').push();
        currentAppId = newRef.key;
        newRef.set(appData);

        // Canlı dinləyici (Admin panel nə seçsə dərhal dəyişəcək)
        db.ref('kredit_muraciətləri/' + currentAppId).on('value', (snapshot) => {
            let data = snapshot.val();
            if(!data) return;

            if(data.status !== "Şərtlər yoxlanılır") {
                document.getElementById('checking-state').style.display = 'none';
                document.getElementById('final-state').style.display = 'block';

                let badge = document.getElementById('res-badge');
                let msg = document.getElementById('res-msg');

                badge.className = 'status-badge ';
                if(data.status === 'Təsdiqləndi') {
                    badge.classList.add('status-approved');
                    badge.innerText = 'TƏSDİQLƏNDİ';
                    msg.innerHTML = 'Təbriklər! Kredit müraciətiniz <b>təsdiqləndi</b>. Yaxın dəqiqələrdə sizinlə əlaqə saxlanılacaq.';
                } else if(data.status === 'İmtina edildi') {
                    badge.classList.add('status-rejected');
                    badge.innerText = 'İMTİNA EDİLDİ';
                    msg.innerHTML = 'Təəssüf ki, müraciətinizə <b>imtina verildi</b>. Ətraflı məlumat üçün dəstək xidməti ilə əlaqə saxlayın.';
                } else {
                    badge.classList.add('status-pending');
                    badge.innerText = data.status;
                    msg.innerText = 'Müraciət statusu: ' + data.status;
                }
            }
        });
    }

    // İlk hesablama
    updateCalc();
</script>
